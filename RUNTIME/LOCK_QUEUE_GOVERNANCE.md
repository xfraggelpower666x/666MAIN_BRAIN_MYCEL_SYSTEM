# LOCK / QUEUE / READBACK GOVERNANCE

LOCK_OWNER=666MAIN BRAIN MYCEL SYSTEM
QUEUE_OWNER=666MAIN BRAIN MYCEL SYSTEM

## Purpose

Prevent two coordination strands from mutating the same canonical target or publishing competing current states.

## Pre-write gate

TARGET
→ AUTHORITY
→ CURRENT_HEAD
→ WRITE_SCOPE
→ CONFLICT_SCAN
→ LOCK_DECISION
→ PRECHANGE_RECOVERY
→ WRITE
→ READBACK
→ RELEASE

## Lock rules

LOCK_REQUIRED_FOR_SHARED_CANONICAL_TARGET=TRUE
LOCK_REQUIRED_FOR_POINTER_PUBLICATION=TRUE
FOREIGN_SYSTEM_LOCK_NE_MAIN_BRAIN_AUTHORITY=TRUE
STALE_LOCK_MUST_REVALIDATE=TRUE
LOCK_WITHOUT_OBSERVED_WRITE_NE_COMPLETION=TRUE

## Queue rules

Queue only writes that cannot safely run concurrently. Reads and independent analyses should not be queued behind unrelated writes.

QUEUE_IS_NOT_PRIORITY_AUTHORITY=TRUE
OWNER_DIRECT_INTENT_MAY_REPRIORITIZE=TRUE
DEPENDENCY_CONSTRAINTS_MAY_REORDER=TRUE

## Readback

WRITE_NE_VERIFIED_UNTIL_READBACK=TRUE
POINTER_WRITE_REQUIRES_FINAL_READBACK=TRUE
READBACK_FAILURE=READBACK_PENDING
